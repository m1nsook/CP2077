cmake_minimum_required(VERSION 3.25)
project(Kineticle VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(kineticle_tests tests/physics.cpp)
  target_include_directories(kineticle_tests PRIVATE src)
  add_test(NAME standalone_mass_policy COMMAND kineticle_tests)
endif()
if(WIN32)
  include(FetchContent)
  set(RED4EXT_HEADER_ONLY ON CACHE BOOL "" FORCE)
  set(RED4EXT_INSTALL OFF CACHE BOOL "" FORCE)
  # Exact SDK layout used by LTBF for patch 2.50.
  FetchContent_Declare(red4ext_sdk GIT_REPOSITORY https://github.com/jackhumbert/RED4ext.SDK.git GIT_TAG 21efd4250f18f6703d6ff7caba0245093b834859)
  FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3)
  FetchContent_MakeAvailable(red4ext_sdk nlohmann_json)
  add_library(Kineticle SHARED src/plugin.cpp)
  target_link_libraries(Kineticle PRIVATE RED4ext::SDK nlohmann_json::nlohmann_json version)
  target_compile_definitions(Kineticle PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(Kineticle PRIVATE /utf-8)
  set_target_properties(Kineticle PROPERTIES PREFIX "" OUTPUT_NAME "Kineticle")
endif()
